\chapter{Metodología de Trabajo: Arquitectura y Planificación}\label{metodologia}
\azul{Este capítulo consolida la estrategia metodológica y la planificación operativa del proyecto. Se establecen las fronteras de la intervención, los objetivos medibles y el desglose exhaustivo de actividades y tareas. La literatura en gestión de proyectos sostiene que una Estructura de Desglose del Trabajo (EDT) bien cimentada constituye el insumo indispensable para proyectar la programación temporal y asegurar el control del alcance \cite[p. ~84]{pmi2021pmbok}.}

\section{Alcance: Delimitación de la Intervención}

\azul{Es necesario definir con precisión los límites del desarrollo. El texto debe precisar las funcionalidades, estructuras o módulos que construirá el sistema y enunciar explícitamente aquellos elementos que quedarán fuera de la intervención. Esta delimitación evita la expansión descontrolada del proyecto durante su ejecución.}

\moradoc{Ejemplo: El sistema de monitoreo de caudales procesará datos hidrológicos mediante sensores telemétricos. La implementación contempla exclusivamente la instalación de hardware en los puntos de captación y el desarrollo de un panel web de visualización. Sin embargo, el diseño excluye la automatización de compuertas físicas de apertura y el mantenimiento de la red hidráulica preexistente.}





\section{Objetivos}\label{objetivos}
\azul{Esta sección enuncia los propósitos medibles del proyecto. Cada objetivo debe iniciar con un verbo en infinitivo que denote una acción cognitiva o práctica evaluable \cite[p. ~37]{hernandez2014metodologia}.}

\subsection{Objetivo General}
\azul{Se declara la meta suprema del desarrollo en un párrafo continuo, integrando el artefacto ingenieril, el contexto de aplicación y la finalidad tecnológica.}
\moradoc{\begin{ejemplo}
		Diseñar e implementar un sistema telemétrico de control de caudales para optimizar la distribución de agua potable en comités de agua rural de la provincia de Osorno.
\end{ejemplo}}



\subsection{Objetivos Específicos}
\azul{Se formulan entre tres y cinco objetivos específicos estructurados según el ciclo de vida CDIO (Concebir, Diseñar, Implementar, Operar) \cite[p. ~34]{crawley2014rethinking}. Se regulan estrictamente en función del número de estudiantes que conforman el grupo de trabajo. Cada uno de estos propósitos debe dividir  directamente en actividades y tareas concretas.}

\begin{itemize}
	\item \azul{\textbf{Proyecto de 1 estudiante:} Basta con formular exactamente 3 objetivos específicos. Es obligatorio cumplir únicamente con las fases de \textbf{Concebir, Diseñar e Implementar (CDI)}.}
	\item \azul{\textbf{Proyecto de 2 estudiantes:} Se deben estructurar entre 3 y 4 objetivos específicos. Debe contemplar las fases de \textbf{Concebir, Diseñar e Implementar (CDI)}, siendo la fase de \textbf{Operar (O)} de carácter opcional.}
	\item \azul{\textbf{Proyecto de 3 estudiantes:} Requiere obligatoriamente la formulación de entre 4 y 5 objetivos específicos. En este escenario, es mandatario cumplir con la totalidad de las fases de la sigla \textbf{CDIO (Concebir, Diseñar, Implementar y Operar)}.}
\end{itemize}

\moradoc{\begin{ejemplo}{Objetivos Específicos (Caso de 3 estudiantes - CDIO Completo)}
\begin{enumerate}
	\item\label{oe1} \textbf{[Concebir]} Diagnosticar los requerimientos técnicos y las condiciones operativas de la red hídrica rural en el sector de estudio.
	\item\label{oe2} \textbf{[Diseñar]} Diseñar la arquitectura del sistema telemétrico, integrando la topología de red de sensores y la estructura de almacenamiento de datos.
	\item\label{oe3} \textbf{[Implementar]} Construir el prototipo funcional mediante la integración de hardware de medición y el desarrollo del software de monitoreo.
	\item\label{oe4} \textbf{[Operar]} Validar el desempeño del sistema a través de pruebas de transmisión y precisión de lecturas en un entorno simulado de operación real.
	\end{enumerate}
\end{ejemplo}
}



\section{Desglose de Actividades y Tareas}\label{s:actividades}
\azul{El desglose de actividades y tareas no se puede omitir bajo ninguna circunstancia. Constituye la descomposición jerárquica del proyecto donde cada objetivo específico se divide en actividades, y cada actividad en tareas operativas atómicas, específicas y desprovistas de ambigüedad. La definición precisa de estas tareas representa el insumo directo para la asignación de recursos, la estimación de duraciones y la posterior construcción de la carta Gantt \cite[p. ~106]{pmi2021pmbok}.}


\begin{ejemplo}{Ejemplo de Desglose Estructurado}~\\
	\moradoc{
\textbf{Objetivo Específico 1 (Concebir): Diagnosticar requerimientos y condiciones operativas.}
\begin{itemize}
	\item \textbf{Actividad 1.1: Levantamiento de información técnica en terreno.}
	\begin{itemize}
		\item \textit{Tarea 1.1.1:} Aplicar pautas de observación directa en tres puntos de captación de agua rural.
		\item \textit{Tarea 1.1.2:} Entrevistar a dos operadores técnicos para identificar fallas recurrentes en la medición manual.
		\item \textit{Tarea 1.1.3:} Tabular los parámetros hidráulicos de presión y caudal máximo registrados históricamente.
	\end{itemize}
	\item \textbf{Actividad 1.2: Definición de especificaciones técnicas del sistema.}
	\begin{itemize}
		\item \textit{Tarea 1.2.1:} Sintetizar la lista de requerimientos funcionales y no funcionales en un documento de especificación.
		\item \textit{Tarea 1.2.2:} Seleccionar los sensores de caudal y módulos de transmisión acordes con las restricciones energéticas del sector.
	\end{itemize}
\end{itemize}}

\moradoc{
\textbf{Objetivo Específico 2 (Diseñar): Diseñar la arquitectura del sistema telemétrico.}
\begin{itemize}
	\item \textbf{Actividad 2.1: Modelado de la infraestructura lógica y física.}
	\begin{itemize}
		\item \textit{Tarea 2.1.1:} Elaborar el diagrama de bloques de la red de sensores y nodos de transmisión.
		\item \textit{Tarea 2.1.2:} Diseñar el esquema de la base de datos relacional en Tercera Forma Normal (3FN).
		\item \textit{Tarea 2.1.3:} Diseñar los prototipos de baja fidelidad para las interfaces del panel de monitoreo web.
	\end{itemize}
\end{itemize}
}
\moradoc{
\textbf{Objetivo Específico 3 (Implementar): Construir el prototipo funcional.}
\begin{itemize}
	\item \textbf{Actividad 3.1: Ensamblaje e integración de hardware.}
	\begin{itemize}
		\item \textit{Tarea 3.1.1:} Programar la placa microcontroladora para la lectura analógica del sensor de flujo.
		\item \textit{Tarea 3.1.2:} Configurar el módulo de comunicación inalámbrica para el envío de paquetes vía protocolo MQTT.
	\end{itemize}
	\item \textbf{Actividad 3.2: Desarrollo de la plataforma de monitoreo.}
	\begin{itemize}
		\item \textit{Tarea 3.2.1:} Construir la API para la recepción y persistencia de datos telemétricos.
		\item \textit{Tarea 3.2.2:} Implementar las vistas de usuario para la visualización de gráficos temporales de caudal.
	\end{itemize}
\end{itemize}
}
\moradoc{
\textbf{Objetivo Específico 4 (Operar - Según corresponda): Validar el desempeño del sistema.}
\begin{itemize}
	\item \textbf{Actividad 4.1: Ejecución de pruebas de estrés y precisión en entorno real o simulado.}
	\begin{itemize}
		\item \textit{Tarea 4.1.1:} Ejecutar 50 ensayos de transmisión continua para medir la tasa de pérdida de paquetes.
		\item \textit{Tarea 4.1.2:} Calibrar los sensores contrastando las lecturas digitales contra un medidor patrón certificado.
		\item \textit{Tarea 4.1.3:} Consolidar los resultados cuantitativos en el informe final de validación.
	\end{itemize}
\end{itemize}
}
\end{ejemplo}
